cmake_minimum_required(VERSION 3.16)
project(falloutquest_nif_tests LANGUAGES CXX)
enable_testing()
add_executable(loading_material_tests loading_material_tests.cpp ../../app/src/main/cpp/npc/fo3-actor-animation.cpp)
target_compile_features(loading_material_tests PRIVATE cxx_std_17)
target_compile_options(loading_material_tests PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
target_include_directories(loading_material_tests PRIVATE . ../../app/src/main/cpp/data ../../app/src/main/cpp/npc)
add_test(NAME loading_material_tests COMMAND loading_material_tests)
add_executable(actor_animation_tests actor_animation_tests.cpp ../../app/src/main/cpp/npc/fo3-actor-animation.cpp)
target_compile_features(actor_animation_tests PRIVATE cxx_std_17)
target_compile_options(actor_animation_tests PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
target_include_directories(actor_animation_tests PRIVATE . ../../app/src/main/cpp/data ../../app/src/main/cpp/npc)
add_test(NAME actor_animation_tests COMMAND actor_animation_tests)
find_package(ZLIB REQUIRED)
add_executable(npc_record_tests npc_record_tests.cpp ../../app/src/main/cpp/npc/fo3-npc.cpp ../../app/src/main/cpp/data/fo3-esm-reader.cpp)
target_compile_features(npc_record_tests PRIVATE cxx_std_17)
target_compile_options(npc_record_tests PRIVATE -Wall -Wextra -Werror)
target_include_directories(npc_record_tests PRIVATE . ../../app/src/main/cpp/data ../../app/src/main/cpp/npc)
target_link_libraries(npc_record_tests PRIVATE ZLIB::ZLIB)
add_test(NAME npc_record_tests COMMAND npc_record_tests)
add_executable(actor_runtime_tests actor_runtime_tests.cpp ../../app/src/main/cpp/npc/fo3-actor-animation.cpp)
target_compile_features(actor_runtime_tests PRIVATE cxx_std_17)
target_compile_options(actor_runtime_tests PRIVATE -Wall -Wextra -Werror)
target_include_directories(actor_runtime_tests PRIVATE ../../app/src/main/cpp/data ../../app/src/main/cpp/npc)
add_test(NAME actor_runtime_tests COMMAND actor_runtime_tests)
