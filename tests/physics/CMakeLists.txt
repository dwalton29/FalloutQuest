cmake_minimum_required(VERSION 3.16)
project(falloutquest_collision_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/FalloutPhysics.cmake)
add_executable(collision_tests collision_tests.cpp)
target_compile_features(collision_tests PRIVATE cxx_std_17)
target_compile_options(collision_tests PRIVATE -Wall -Wextra -UNDEBUG -Wno-unused-function)
target_include_directories(collision_tests PRIVATE mock ../../app/src/main/cpp ../../app/src/main/cpp/world ../../app/src/main/cpp/data ../../app/src/main/cpp/physics)
target_link_libraries(collision_tests PRIVATE Threads::Threads fq_physics)
add_test(NAME collision_tests COMMAND collision_tests)
add_executable(rigid_world_tests rigid_world_tests.cpp)
target_compile_features(rigid_world_tests PRIVATE cxx_std_17)
target_compile_options(rigid_world_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(rigid_world_tests PRIVATE fq_physics)
add_test(NAME rigid_world_tests COMMAND rigid_world_tests)

add_executable(grip_response_tests grip_response_tests.cpp)
target_compile_features(grip_response_tests PRIVATE cxx_std_17)
target_compile_options(grip_response_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(grip_response_tests PRIVATE fq_physics)
add_test(NAME grip_response_tests COMMAND grip_response_tests)

add_executable(ragdoll_tests ragdoll_tests.cpp ../../app/src/main/cpp/npc/fo3-actor-animation.cpp)
target_compile_features(ragdoll_tests PRIVATE cxx_std_17)
target_compile_options(ragdoll_tests PRIVATE -Wall -Wextra -Werror -Wno-unused-function -UNDEBUG)
target_include_directories(ragdoll_tests PRIVATE ../nif ../../app/src/main/cpp/data ../../app/src/main/cpp/npc)
target_link_libraries(ragdoll_tests PRIVATE fq_physics)
add_test(NAME ragdoll_tests COMMAND ragdoll_tests)
